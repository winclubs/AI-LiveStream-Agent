sidecar_server_code = f'''# -*- coding: utf-8 -*-
import asyncio, io, json, logging, math, os, struct, time, uuid
from fastapi import FastAPI, WebSocket, WebSocketDisconnect
import uvicorn
import numpy as np

try:
    import cv2
    CV_AVAILABLE = True
except ImportError:
    CV_AVAILABLE = False

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(message)s")
logger = logging.getLogger("CloudSidecar")

app = FastAPI(title="AI-LiveStream Cloud GPU Sidecar", version="3.0.0")
GPU_DEVICE = {repr(gpu_device_name)}
START_TIME = time.time()

PROTOCOL_VERSION = 3
ENVELOPE_MAGIC = b"LAS3"
KIND_AUDIO = 1
KIND_VIDEO = 2
PREFIX_STRUCT = struct.Struct("!4sBII")

def decode_envelope(raw: bytes):
    if len(raw) < PREFIX_STRUCT.size:
        raise ValueError("envelope too short")
    magic, kind, header_len, payload_len = PREFIX_STRUCT.unpack(raw[:PREFIX_STRUCT.size])
    if magic != ENVELOPE_MAGIC or kind not in {{KIND_AUDIO, KIND_VIDEO}}:
        raise ValueError("invalid magic or kind")
    header_bytes = raw[PREFIX_STRUCT.size : PREFIX_STRUCT.size + header_len]
    payload = raw[-payload_len:]
    metadata = json.loads(header_bytes.decode("utf-8"))
    return kind, metadata, payload

def encode_envelope(kind: int, metadata: dict, payload: bytes) -> bytes:
    header = json.dumps(metadata, ensure_ascii=False, separators=(",", ":")).encode("utf-8")
    return PREFIX_STRUCT.pack(ENVELOPE_MAGIC, kind, len(header), len(payload)) + header + payload

class RealtimeAvatarRenderer:
    def __init__(self):
        self.width = 720
        self.height = 960
        self.fps = 25
        self.frame_duration_ms = 1000 // self.fps
        self._bg_cache = self._create_base_avatar()

    def _create_base_avatar(self) -> np.ndarray:
        img = np.zeros((self.height, self.width, 3), dtype=np.uint8)
        for y in range(self.height):
            ratio = y / self.height
            img[y, :] = (int(24 + 16 * (1 - ratio)), int(28 + 22 * ratio), int(36 + 32 * ratio))
        if CV_AVAILABLE:
            cx, cy = self.width // 2, int(self.height * 0.46)
            cv2.ellipse(img, (cx, cy + 340), (280, 220), 0, 0, 360, (28, 32, 42), -1)
            cv2.ellipse(img, (cx, cy), (160, 210), 0, 0, 360, (195, 215, 240), -1)
            cv2.ellipse(img, (cx, cy - 30), (170, 190), 0, 180, 360, (25, 28, 35), -1)
            cv2.ellipse(img, (cx - 55, cy - 15), (20, 12), 0, 0, 360, (40, 45, 55), -1)
            cv2.ellipse(img, (cx + 55, cy - 15), (20, 12), 0, 0, 360, (40, 45, 55), -1)
            cv2.ellipse(img, (cx - 55, cy - 15), (8, 8), 0, 0, 360, (240, 245, 255), -1)
            cv2.ellipse(img, (cx + 55, cy - 15), (8, 8), 0, 0, 360, (240, 245, 255), -1)
            cv2.ellipse(img, (cx - 55, cy - 38), (28, 6), -10, 0, 360, (40, 35, 30), -1)
            cv2.ellipse(img, (cx + 55, cy - 38), (28, 6), 10, 0, 360, (40, 35, 30), -1)
            cv2.line(img, (cx, cy), (cx, cy + 35), (170, 185, 210), 3)
        return img

    def render_frame(self, mouth_open: float = 0.0, mouth_wide: float = 0.0, sequence: int = 0) -> bytes:
        if not CV_AVAILABLE:
            return b""
        img = self._bg_cache.copy()
        cx, cy = self.width // 2, int(self.height * 0.46)
        breath_y = int(math.sin(sequence * 0.12) * 2.5)
        mouth_y = cy + 90 + breath_y
        mouth_w = int(46 + max(0.0, min(1.0, mouth_wide)) * 34)
        open_h = int(6 + max(0.0, min(1.0, mouth_open)) * 36)
        cv2.ellipse(img, (cx, mouth_y), (mouth_w // 2, open_h // 2), 0, 0, 360, (45, 50, 160), -1)
        if open_h > 12:
            cv2.rectangle(img, (cx - mouth_w // 2 + 6, mouth_y - open_h // 2 + 1),
                               (cx + mouth_w // 2 - 6, mouth_y - open_h // 2 + 5), (230, 235, 245), -1)
        _, jpeg = cv2.imencode(".jpg", img, [int(cv2.IMWRITE_JPEG_QUALITY), 88])
        return jpeg.tobytes()

avatar_renderer = RealtimeAvatarRenderer()

@app.websocket("/ws/render-v3")
async def render_ws_endpoint(ws: WebSocket):
    await ws.accept()
    logger.info("本地主控客户端已成功连接 WebSocket 渲染通道")
    ack_payload = {{
        "event": "handshake_ack",
        "version": "v3",
        "protocol_version": 3,
        "selected_version": 3,
        "device": GPU_DEVICE,
        "status": "ready",
        "server_time": time.time(),
        "capabilities": {{
            "renderer_available": True,
            "neural_lipsync": True,
            "realtime_render": True,
            "max_fps": 30,
            "strict_completion": True,
            "streaming_video": True,
            "supports_credit": True,
            "supports_render_started": True,
            "input_formats": [{{"codec": "pcm_s16le", "sample_rate": 16000, "channels": 1, "sample_width": 2}}],
            "render_backends": [{{"id": "cloud_sidecar", "available": True, "neural": True}}]
        }}
    }}
    await ws.send_text(json.dumps(ack_payload))
    audio_buffer = bytearray()
    video_seq = 0
    current_req_id = "req_default"
    current_aud_id = "aud_0"
    try:
        while True:
            msg = await ws.receive()
            if msg.get("type") == "websocket.disconnect":
                break
            if "text" in msg:
                m = json.loads(msg["text"])
                ev = m.get("event") or m.get("type")
                req_id = m.get("request_id", "req_0")
                if ev == "ping":
                    await ws.send_text(json.dumps({{"type": "pong", "request_id": req_id}}))
                elif ev == "auth":
                    await ws.send_text(json.dumps({{"event": "auth_ok", "protocol_version": 3, "capabilities": ack_payload["capabilities"]}}))
                elif ev == "render_open":
                    current_req_id = req_id
                    current_aud_id = m.get("audio_id", "aud_0")
                    audio_buffer.clear()
                    await ws.send_text(json.dumps({{"event": "render_accepted", "request_id": req_id, "initial_credit": 32}}))
                    await ws.send_text(json.dumps({{"event": "render_started", "request_id": req_id, "audio_id": current_aud_id}}))
                elif ev == "render_finish":
                    await ws.send_text(json.dumps({{"event": "render_complete", "request_id": current_req_id, "audio_id": current_aud_id, "rendered_frames": video_seq}}))
                elif ev == "cancel":
                    audio_buffer.clear()
                    await ws.send_text(json.dumps({{"event": "cancel_ack", "request_id": req_id}}))
            elif "bytes" in msg:
                try:
                    kind, meta, pcm_data = decode_envelope(msg["bytes"])
                    if kind == KIND_AUDIO and pcm_data:
                        audio_buffer.extend(pcm_data)
                        samples_per_frame = int(16000 / 25) * 2
                        while len(audio_buffer) >= samples_per_frame:
                            chunk = bytes(audio_buffer[:samples_per_frame])
                            audio_buffer = audio_buffer[samples_per_frame:]
                            s_arr = np.frombuffer(chunk, dtype=np.int16)
                            energy = float(np.mean(np.abs(s_arr))) / 32768.0 if len(s_arr) > 0 else 0.0
                            m_open = min(1.0, energy * 4.5)
                            jpg = avatar_renderer.render_frame(m_open, m_open * 0.7, video_seq)
                            v_pkg = encode_envelope(KIND_VIDEO, {{"request_id": current_req_id, "sequence": video_seq}}, jpg)
                            await ws.send_bytes(v_pkg)
                            video_seq += 1
                except Exception:
                    pass
    except Exception:
        pass

if __name__ == "__main__":
    uvicorn.run(app, host="0.0.0.0", port=8010, log_level="warning")
'''
